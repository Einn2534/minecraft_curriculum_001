player.onChat("q27r03", function () {
    agent.teleport(pos(-6, 0, 9), SOUTH)
    let legs = [[0, 5], [12, 0], [0, 2]]
    for (let leg of legs) {
        let direction = SOUTH
        if (leg[0] > 0) {
            direction = EAST
        } else if (leg[0] < 0) {
            direction = WEST
        } else if (leg[1] < 0) {
            direction = NORTH
        }
        agent.teleport(agent.getPosition(), direction)
        agent.move(FORWARD, Math.abs(leg[0]) + Math.abs(leg[1]))
    }
})
