function check_q08r03_6(): number {
    let repaired = 0
    let points = [pos(-4, 2, 12), pos(-2, 2, 12), pos(0, 2, 12), pos(2, 2, 12), pos(4, 2, 12)]
    for (let point of points) {
        if (!blocks.testForBlock(GLOWSTONE, point)) {
            blocks.place(GLOWSTONE, point)
            repaired += 1
        }
    }
    return repaired
}
